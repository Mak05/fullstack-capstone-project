name: CI/CD

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4
      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: Install backend dependencies
        working-directory: project/blob/main/backend
        run: npm install
      - name: Install frontend dependencies
        working-directory: project/blob/main/frontend
        run: npm install
      - name: Build frontend
        working-directory: project/blob/main/frontend
        run: npm run build
      - name: Start backend
        working-directory: project/blob/main/backend
        run: npm run start &
      - name: Smoke test backend
        run: curl -f http://localhost:3000/api/gifts
      - name: Deploy status
        run: echo "CI/CD workflow completed successfully"

Result: SUCCESS
